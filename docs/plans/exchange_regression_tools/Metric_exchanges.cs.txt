using System.Globalization;
using MiniRing;
using MiniRing.Bots;
// args: pairs seeds seconds tag [set]   pair token: style name, or @id (roster character), @id~ (roster, GamePlan off)
var pairs = args[0].Split(',').Select(p => p.Split(':')).ToList();
var seeds = args[1].Split(',').Select(int.Parse).ToList();
var seconds = float.Parse(args[2], CultureInfo.InvariantCulture);
var tag = args[3];
var set = args.Length > 4 ? args[4] : "";
var tuning = new FighterTuning();
var ci = CultureInfo.InvariantCulture;
var bins = new[] { 0.5f, 0.75f, 1.0f, 1.25f, 1.5f, 2.0f };
Contender Make(string tok, string? foe)
{
    if (!tok.StartsWith('@')) return Contender.Named(tok);
#if ROSTER
    var tilde = tok.Contains('~'); var id = tok.TrimStart('@').Split('~', '^')[0];
    var style = MiniRing.Characters.CharacterRoster.Get(id).Resolve(foe?.TrimStart('@').Split('~', '^')[0]);
    if (tilde) style = style with { GamePlan = null };
#if RHYTHM
    if (tok.Contains('^')) style = style with { Rhythm = null };
#endif
    return Contender.Of(style);
#else
    throw new Exception("roster not compiled");
#endif
}
double[] tot = new double[40]; int nb = 0; var hist = new double[2, bins.Length + 1]; double[] stand = new double[2];
foreach (var p in pairs) foreach (var seed in seeds)
{
    var red = Make(p[0], p[1]); var blue = Make(p[1], p[0]);
    var rb = new List<IFighterControl>(); var bb = new List<IFighterControl>();
    var corners = new BoxingCorners(blue.Name, s => { var c = blue.Create(s); bb.Add(c); return c; }, s => { var c = red.Create(s); rb.Add(c); return c; }, PlayerName: red.Name, PlayerBody: red.Build, CpuBody: blue.Build, PlayerFace: red.Face, CpuFace: blue.Face, Referee: true);
    using var game = new BoxingGame(tuning, seed, corners);
    nb++;
    int Throws(List<IFighterControl> l) => l.OfType<BoxerBot>().Sum(b => b.Stats.Techniques.Values.Sum(t => t.Thrown));
    int Def(List<IFighterControl> l, params Defense[] ds) => l.OfType<BoxerBot>().Sum(b => ds.Sum(d => b.Stats.Defenses.TryGetValue(d, out var c) ? c.Attempts : 0));
    var ev = new[] { Defense.Duck, Defense.Weave, Defense.Slip, Defense.Sway };
    var st = new[] { Defense.StepBack, Defense.SideStep, Defense.SlipIn, Defense.Roll };
    int prevR = 0, prevB = 0; var tr = new List<float>(); var tb = new List<float>();
    var steps = (int)(seconds * 60);
    float dmgR = 0, dmgB = 0; int landR = 0, landB = 0;
    for (var i = 0; i < steps; i++)
    {
        var t = i / 60f;
        game.Step(default, 1f / 60f);
        foreach (var h in game.RecentHits) { if (h.Attacker == game.Player) { dmgR += h.Damage; landR++; } else { dmgB += h.Damage; landB++; } }
        var a = Throws(rb); var b = Throws(bb);
        if (a > prevR) tr.Add(t); if (b > prevB) tb.Add(t); prevR = a; prevB = b;
        if (!game.Resting && game.Player.State == FighterState.Standing && game.Cpu.State == FighterState.Standing)
        {
            var d = game.Player.ChestPosition - game.Cpu.ChestPosition; var dist = MathF.Sqrt((d.X * d.X) + (d.Z * d.Z)) / ((game.Player.ArmReach + game.Cpu.ArmReach) * 0.5f);
            var k = 0; while (k < bins.Length && dist >= bins[k]) k++;
            hist[0, k] += 1 / 60.0; stand[0] += 1 / 60.0; tot[30] += dist / 60.0;
        }
    }
    // exchanges: cluster all throws with gap <= 1 s; cluster with both sides = exchange
    var all = tr.Select(x => (x, 0)).Concat(tb.Select(x => (x, 1))).OrderBy(x => x.Item1).ToList();
    int ex = 0; for (int i = 0; i < all.Count;) { int j = i; bool r0 = false, b0 = false; while (j < all.Count && (j == i || all[j].Item1 - all[j - 1].Item1 <= 1f)) { if (all[j].Item2 == 0) r0 = true; else b0 = true; j++; } if (r0 && b0) ex++; i = j; }
    tot[0] += prevR; tot[1] += prevB; tot[2] += landR; tot[3] += landB; tot[4] += dmgR; tot[5] += dmgB;
    tot[6] += Def(rb, ev); tot[7] += Def(bb, ev); tot[8] += Def(rb, st); tot[9] += Def(bb, st);
    tot[10] += Def(rb, Defense.Block); tot[11] += Def(bb, Defense.Block); tot[12] += Def(rb, Defense.Parry); tot[13] += Def(bb, Defense.Parry);
    tot[16] += rb.OfType<BoxerBot>().Sum(b => ev.Sum(d => b.Stats.Defenses.TryGetValue(d, out var c) ? c.Failed : 0)); tot[17] += bb.OfType<BoxerBot>().Sum(b => ev.Sum(d => b.Stats.Defenses.TryGetValue(d, out var c) ? c.Failed : 0));
    tot[14] += ex; for (int i = 1; i < all.Count; i++) if (all[i].Item2 != all[i - 1].Item2 && all[i].Item1 - all[i - 1].Item1 <= 1f) tot[15]++;
}
string Hs() { var s = ""; var lo = 0f; for (var k = 0; k <= bins.Length; k++) { s += string.Create(ci, $" <{(k < bins.Length ? bins[k] : 99):0.##}:{100 * hist[0, k] / Math.Max(1e-9, stand[0]):0}%"); } return s; }
Console.WriteLine(string.Create(ci, $"{tag}\t{args[0]}\tbouts={nb}\tthrown/bout R={tot[0] / nb:0.0} B={tot[1] / nb:0.0}\tlanded R={tot[2] / nb:0.0} B={tot[3] / nb:0.0}\tdmg R={tot[4] / nb:0} B={tot[5] / nb:0}\tevasions R={tot[6] / nb:0.0} B={tot[7] / nb:0.0} (failed R={tot[16] / nb:0.0} B={tot[17] / nb:0.0})\tsteps R={tot[8] / nb:0.0} B={tot[9] / nb:0.0}\tblocks R={tot[10] / nb:0.0} B={tot[11] / nb:0.0}\tparry R={tot[12] / nb:0.0} B={tot[13] / nb:0.0}\texchangeClusters={tot[14] / nb:0.0} alternations={tot[15] / nb:0.0}\tmeanDist/reach={tot[30] / Math.Max(1e-9, stand[0]):0.00}\tstandSec={stand[0] / nb:0}\tdistHist(chest dist/arm reach){Hs()}"));
